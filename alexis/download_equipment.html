const axios = require('axios'); const fs = require('fs').promises; const path = require('path'); // URL de la solicitud const url = 'http://10.1.64.119:6060/api/v1/cube/get/dimension/?f=equipment'; // Ruta y nombre del archivo donde se guardará el resultado const outputFile = path.join('output', 'equipment_data.json'); async function fetchAndSaveData() { try { // Crear la carpeta 'output' si no existe await fs.mkdir(path.dirname(outputFile), { recursive: true }); // Realizar la solicitud GET const response = await axios.get(url); // Verificar si la solicitud fue exitosa if (response.status === 200) { // Convertir los datos a formato JSON con formato legible const data = JSON.stringify(response.data, null, 2); // Guardar los datos en el archivo await fs.writeFile(outputFile, data); console.log(`Datos guardados exitosamente en: ${outputFile}`); } else { console.error(`Error al obtener los datos. Código de estado: ${response.status}`); } } catch (error) { if (error.response) { console.error(`Error en la solicitud: ${error.response.status} - ${error.response.statusText}`); } else if (error.code === 'ENOENT' || error.code === 'EACCES') { console.error('Error de permisos o directorio no accesible.'); } else { console.error(`Error inesperado: ${error.message}`); } } } // Ejecutar la función fetchAndSaveData();
